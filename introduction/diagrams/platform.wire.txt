legend:
client   = Clients | apps, users, services {client}
traefik  = Traefik | Gateway API {service}
authgw   = Auth Gateway | token + edge authz {service}
pdp      = Cerbos PDP | generated policies {service}
idp      = OIDC provider | Keycloak or another issuer {external}
pg       = Payment Gateway {service}
co       = Checkout | API + UI {service}
af       = AuditFlow {service}
ui       = Customer Portal {service}
rp       = Reporting | read API {service}
postgres = PostgreSQL | database per service {store}
redis    = Redis / Valkey | idempotency {store}
rabbit   = Message broker | AuditFlow buffer {queue}
sinks    = Audit stores | analytics and archive destinations {external}

traefik -> af
co.right -> postgres.left
pg.right:1 -> redis.left
pg.top -> postgres.left
pg.bottom ..> af.top : audit events

af.right -> sinks.left : deliver
traefik -> rp
rp.right ..> sinks.left : read
